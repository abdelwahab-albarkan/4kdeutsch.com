import { ImageResponse } from 'next/og';
import fs from 'fs';
import path from 'path';

// Branded default social-share card; the backdrop is the library image public/images/iptv/social/4kdeutsch-streaming.jpg.
export const alt = '4kdeutsch — Premium IPTV-Abonnement in Deutschland';
export const size = { width: 1200, height: 630 };
export const contentType = 'image/png';

function backdrop(): string | null {
  try {
    const buf = fs.readFileSync(path.join(process.cwd(), 'public/images/iptv/social/4kdeutsch-streaming.jpg'));
    return `data:image/jpeg;base64,${buf.toString('base64')}`;
  } catch {
    return null;
  }
}

        )}
